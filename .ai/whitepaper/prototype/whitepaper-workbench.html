SELECT
  DATE_FORMAT(t.case_time, '%Y-%m') AS case_month,
  ct.case_type_name,
  COUNT(t.case_no) AS case_count,
  COUNT(DISTINCT t.suspect_id) AS suspect_count
FROM dwd_case_suspect t
LEFT JOIN dim_case_type ct ON t.case_type_code = ct.case_type_code
LEFT JOIN dim_org_area org ON t.org_code = org.org_code
WHERE t.case_time >= :start_time
  AND t.case_status = :case_status
  AND org.area_name LIKE CONCAT('%', :area_name, '%')
GROUP BY DATE_FORMAT(t.case_time, '%Y-%m'), ct.case_type_name
ORDER BY case_month ASC
LIMIT 500;
